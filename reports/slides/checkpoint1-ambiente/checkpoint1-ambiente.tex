\setLayout{vertical}
\begin{frame}{Problemas encontrados e resolvidos}

    \begin{alertblock}{O OpenCV 5 removeu o Viola-Jones}
        O \texttt{cv2.CascadeClassifier} não existe mais na versão 5.0.
        Fixamos \texttt{opencv-python==4.14.0.94}.
    \end{alertblock}
    \vspace{0.8em}
    \begin{alertblock}{O grad-cam quebra o \texttt{cv2}}
        Ele instala o \texttt{opencv-python-headless}, que conflita com o \texttt{opencv-python}.
        Removemos o headless depois da instalação.
    \end{alertblock}
    \vspace{0.8em}
    \begin{recommendationblock}{Para a frente B}
        O Viola-Jones do CK+ depende do OpenCV 4: usar o mesmo \texttt{requirements.txt}.
    \end{recommendationblock}

\end{frame}

%--------------------------------------------------------
\begin{frame}[fragile]{Como reproduzir o ambiente}

    \begin{lstlisting}[language=bash, basicstyle=\scriptsize\ttfamily, breaklines=true]
python -m venv .venv
.venv\Scripts\activate
pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cu130
pip install -r requirements.txt
pip uninstall -y opencv-python-headless opencv-python
pip install opencv-python==4.14.0.94
    \end{lstlisting}

    \vspace{0.3em}
    {\footnotesize Os passos e as versões testadas estão no \texttt{requirements.txt} do repositório.}
